# Ледяная Корона: без неё тепло тает, с ней возвращается и не наступает иней
/gamemode creative
/effect clear @s
/gamerule spawn_monsters false
/execute in celestial:frozen_halls run tp @s 15 140 15 0 30
fly
wait 80
/execute in celestial:frozen_halls run fill 10 139 10 20 139 20 celestial:frost_stone_bricks
/execute in celestial:frozen_halls run fill 10 140 10 20 144 20 minecraft:air
/execute in celestial:frozen_halls run tp @s 15 140 15 0 30
/weather clear
/clear @s
/gamemode survival
/effect give @s minecraft:resistance 900 4 true
wait 400
/celestial status
/item replace entity @s armor.head with celestial:ice_crown
wait 300
/say TEST C1 crown_warms (тепло во втором статусе выше первого)
/celestial status
/data get entity @s TicksFrozen
quit
