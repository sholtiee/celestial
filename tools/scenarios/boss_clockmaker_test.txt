/data merge entity @e[type=celestial:clockmaker,limit=1] {Health:500.0f}
wait 10
/say TEST K3 phase2 expect=pass

/execute store result score #sp0 t run attribute @s minecraft:movement_speed get 1000
wait 268
/say TEST K3b beetles_summoned expect=pass
/execute if entity @e[type=celestial:winder_beetle,tag=celestial_no_repel]

wait 20
/say TEST K6 phase3 expect=pass
/execute if entity @e[type=celestial:clockmaker,nbt={Phase:3}]
/say TEST K7 pendulums_rewound expect=pass
/execute if block 32006 163 31994 celestial:swift_pendulum[swinging=true]
/damage @e[type=celestial:clockmaker,limit=1] 400 minecraft:player_attack by @s

/execute store result score #hp t run data get entity @e[type=celestial:clockmaker,limit=1] Health

/say TEST K8 held_by_pendulums expect=pass
/execute if score #hp t matches 119..
