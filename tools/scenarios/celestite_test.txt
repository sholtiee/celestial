# Целестит и Астролябия: руда в породе Астрала, Астролябия наводится на святилище хора и алтарь, полный комплект снимает Тьму
/gamemode creative
/effect clear @s
/clear @s
/execute in celestial:astral run tp @s 0 200 0 0 30
fly
wait 200
/say TEST C1 ore_generated (число заменённых блоков > 0)
/fill -16 60 -16 15 91 15 minecraft:stone replace celestial:celestite_ore
/fill -16 92 -16 15 123 15 minecraft:stone replace celestial:celestite_ore
/fill -16 124 -16 15 155 15 minecraft:stone replace celestial:celestite_ore
/say TEST A1 astrolabe_shrine expect=pass
/gamemode survival
/item replace entity @s weapon.mainhand with celestial:astrolabe
wait 5
useitem
wait 10
/execute if entity @p[nbt={SelectedItem:{components:{"minecraft:lodestone_tracker":{}}}}]
togglehud
shot astrolabe_hand
togglehud
/say TEST S1 celestite_set_darkness expect=pass
/item replace entity @s armor.head with celestial:celestite_helmet
/item replace entity @s armor.chest with celestial:celestite_chestplate
/item replace entity @s armor.legs with celestial:celestite_leggings
/item replace entity @s armor.feet with celestial:celestite_boots
/effect give @s minecraft:darkness 30 0
wait 45
/execute unless entity @p[nbt={active_effects:[{id:"minecraft:darkness"}]}]
togglehud
camera front
wait 5
shot celestite_armor
camera first
togglehud
/clear @s
/gamemode creative
quit
