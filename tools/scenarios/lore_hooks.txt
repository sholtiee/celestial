# S2.1: подвиги открывают листы Летописи: Пожиратель Света → «Аваддон», Морозный Архонт → «Правители сфер», Серафим → слово «Серафим». Проверка: после победы /celestial lore unlock даёт «Открыто листов: 0»
/gamemode creative
/effect clear @s
/gamerule spawn_monsters false
/time set 6000
/gamerule advance_time false
/celestial lore lock all
# Серафим
goto celestial:citadel 0 -20 -8 0 5
wait 120
/gamemode survival
/effect give @s minecraft:resistance 9000 4 true
/effect give @s minecraft:saturation 9000 4 true
/execute at @s run fill ~-3 ~-1 ~2 ~3 ~-1 ~12 celestial:skystone_bricks
/execute at @s run setblock ~ ~ ~8 celestial:seraph_seal
wait 60
/damage @e[type=celestial:fallen_seraph,limit=1] 400 minecraft:player_attack by @s
wait 10
/damage @e[type=celestial:fallen_seraph,limit=1] 400 minecraft:player_attack by @s
wait 10
wait 60
/say TEST H1 seraph_gloss expect=0 (слово уже открыто победой)
/celestial lore gloss seraph
