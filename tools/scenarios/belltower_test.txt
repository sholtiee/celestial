# мелодия: [3, 2, 1, 3, 3]
/gamemode creative
/effect clear @s
/clear @s
/gamerule advance_time false
/gamerule spawn_monsters false
/weather clear
/execute in celestial:chronograd run tp @s 5005 140 5020

/fill 4995 94 4995 5015 98 5015 minecraft:stone
/fill 4995 99 4995 5015 120 5015 minecraft:air
/fill 4995 121 4995 5015 142 5015 minecraft:air

/tp @s 5005.5 128 5008.5 180 25

# L1 подъём по стремянке пешком (держим «вперёд» в лестницу)
/say TEST L1 ladder_climb expect=pass
/gamemode survival
/effect give @s minecraft:resistance 900 4 true

wait 10

wait 265

/execute if entity @p[y=126,dy=3]

/execute if block 5007 126 5002 celestial:sealed_door

wait 8

wait 8

wait 8
